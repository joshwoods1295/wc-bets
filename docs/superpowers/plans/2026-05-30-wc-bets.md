# WC Bets Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scrape granular per-player league stats into a SQLite database, map players to World Cup 2026 national squads, and surface bettable positional matchup flags on a Streamlit dashboard.

**Architecture:** Four decoupled layers — (1) scrapers (FBref via `soccerdata`, plus WC squads/fixtures with a manual-import fallback), (2) a SQLite database accessed through a thin repository module, (3) a pure-Python analysis engine that normalises stats to per-90, computes positional percentiles, and emits matchup flags, and (4) a Streamlit dashboard. The analysis engine is network-free and fully unit-tested; scrapers are tested against saved sample data so CI never hits the network.

**Tech Stack:** Python 3.11+ (venv), `soccerdata`, `pandas`, `sqlite3` (stdlib), `streamlit`, `pytest`.

"""The only module that talks to SQLite."""
import sqlite3
from pathlib import Path

_SCHEMA = Path(__file__).with_name("schema.sql")


def connect(db_path) -> sqlite3.Connection:
    db_path = Path(db_path)
    db_path.parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(db_path)
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def init_db(conn: sqlite3.Connection) -> None:
    conn.executescript(_SCHEMA.read_text())
    conn.commit()


def _upsert(conn, table, row, pk_cols):
    cols = list(row)
    placeholders = ", ".join("?" for _ in cols)
    updates = ", ".join(f"{c}=excluded.{c}" for c in cols if c not in pk_cols)
    sql = (f"INSERT INTO {table} ({', '.join(cols)}) VALUES ({placeholders}) "
           f"ON CONFLICT({', '.join(pk_cols)}) DO UPDATE SET {updates}")
    conn.execute(sql, [row[c] for c in cols])
    conn.commit()


def upsert_player(conn, row):
    _upsert(conn, "players", row, ["player_id"])


def upsert_player_stats(conn, row):
    _upsert(conn, "player_stats", row, ["player_id", "season"])


def upsert_squad_member(conn, player_id, country):
    _upsert(conn, "national_squads",
            {"player_id": player_id, "country": country},
            ["player_id", "country"])


def upsert_fixture(conn, row):
    _upsert(conn, "fixtures", row, ["match_id"])
```

"""Per-90 normalisation. Pure functions, no I/O."""


def per90(total, minutes):
    if not minutes:
        return 0.0
    return round(total * 90.0 / minutes, 4)


def per90_row(row, stat_keys):
    out = dict(row)
    minutes = row.get("minutes", 0)
    for key in stat_keys:
        out[f"{key}_p90"] = per90(row.get(key, 0), minutes)
    return out
```

"""Rank a player's stat against positional peers. Pure functions."""


def percentile_rank(value, peers):
    """Fraction of peers with value <= the given value (0..1)."""
    if not peers:
        return 0.0
    at_or_below = sum(1 for p in peers if p <= value)
    return round(at_or_below / len(peers), 4)
```

"""Matchup flag engine. Pure functions over plain dicts.

A 'rule' pairs a defender stat with an attacker stat. A flag fires when BOTH
players sit at or above the percentile threshold for their respective stat,
relative to positional peers supplied by the caller.
"""
from wcbets.analysis.percentiles import percentile_rank

# (label, defender_stat, attacker_stat)
RULES = [
    ("High-fouling defender vs high fouls-drawn attacker",
     "fouls_p90", "fouls_drawn_p90"),
]


def evaluate_matchup(defender, attacker, peers, threshold=0.80):
    flags = []
    for label, def_stat, atk_stat in RULES:
        def_val = defender.get(def_stat, 0.0)
        atk_val = attacker.get(atk_stat, 0.0)
        def_pct = percentile_rank(def_val, peers.get(f"defender_{def_stat}", []))
        atk_pct = percentile_rank(atk_val, peers.get(f"attacker_{atk_stat}", []))
        if def_pct >= threshold and atk_pct >= threshold:
            flags.append({
                "label": label,
                "defender": defender.get("name"),
                "attacker": attacker.get("name"),
                "defender_stat": def_stat,
                "attacker_stat": atk_stat,
                "defender_value": def_val,
                "attacker_value": atk_val,
                "defender_pct": def_pct,
                "attacker_pct": atk_pct,
            })
    return flags
```

"""Scrape per-player season stats from FBref via soccerdata.

`map_fbref_row` (pure, tested) converts one FBref-shaped record to our schema.
`scrape_league` performs the live fetch and is exercised manually in Step 5.
"""
from wcbets.config import LEAGUES, SEASON
from wcbets.db import repository as repo

# Our canonical name -> the FBref column name we read it from.
FBREF_MAP = {
    "minutes": "Min", "shots": "Sh", "shots_on_target": "SoT",
    "goals": "Gls", "assists": "Ast", "fouls": "Fls", "fouls_drawn": "Fld",
    "yellows": "CrdY", "reds": "CrdR", "offsides": "Off",
    "tackles": "Tkl", "corners_taken": "CK", "saves": "Saves",
}


def map_fbref_row(row, league, season):
    out = {
        "player_id": f"{league}:{row.get('player')}",  # provisional id; see Task 9
        "name": row.get("player"),
        "position": row.get("pos"),
        "club": row.get("team"),
        "league": league,
        "age": _to_int(row.get("age")),
        "season": season,
    }
    for canonical, fbref_col in FBREF_MAP.items():
        out[canonical] = _to_num(row.get(fbref_col, 0))
    return out


def _to_num(v):
    try:
        return float(v) if v not in (None, "") else 0.0
    except (TypeError, ValueError):
        return 0.0


def _to_int(v):
    try:
        return int(str(v).split("-")[0])  # FBref age like "25-031"
    except (TypeError, ValueError):
        return None


def scrape_league(conn, league, season):
    """Live fetch via soccerdata; writes players + stats. Run manually."""
    import soccerdata as sd
    fbref = sd.FBref(leagues=league, seasons=season)
    df = fbref.read_player_season_stats(stat_type="standard").reset_index()
    df.columns = [c if isinstance(c, str) else c[-1] for c in df.columns]
    records = df.to_dict("records")
    for rec in records:
        mapped = map_fbref_row(rec, league, season)
        repo.upsert_player(conn, {k: mapped[k] for k in
            ("player_id", "name", "position", "club", "league", "age")})
        repo.upsert_player_stats(conn, {k: mapped[k] for k in
            (["player_id", "season"] + list(FBREF_MAP))})
    return len(records)


def scrape_all(conn):
    total = 0
    for lg in LEAGUES:
        total += scrape_league(conn, lg, SEASON)
    return total
```

"""Scrape WC 2026 squads + fixtures; reconcile players to league data.

Network functions (`scrape_squads`, `scrape_fixtures`) are run manually; the
reconciliation helpers are pure and unit-tested. If scraping is unavailable or
incomplete, use wcbets.scrape.manual_import instead.
"""
import unicodedata
from wcbets.db import repository as repo


def normalise_name(name):
    nfkd = unicodedata.normalize("NFKD", name or "")
    no_accents = "".join(c for c in nfkd if not unicodedata.combining(c))
    return no_accents.casefold().strip()


def match_player(norm_name, players):
    for p in players:
        if normalise_name(p["name"]) == norm_name:
            return p["player_id"]
    return None


def scrape_squads(conn, country, fbref_team_id):
    """Live: read a national team's player list from FBref and link to players.

    Reads existing players from the DB, matches squad members by normalised
    name, and inserts national_squads rows. Returns count linked.
    """
    import soccerdata as sd
    fbref = sd.FBref(leagues=f"INT-World Cup", seasons="2026")
    df = fbref.read_player_season_stats(stat_type="standard").reset_index()
    df.columns = [c if isinstance(c, str) else c[-1] for c in df.columns]
    existing = [dict(player_id=r[0], name=r[1]) for r in
                conn.execute("SELECT player_id, name FROM players")]
    linked = 0
    for rec in df.to_dict("records"):
        if rec.get("team") != country:
            continue
        pid = match_player(normalise_name(rec.get("player", "")), existing)
        if pid:
            repo.upsert_squad_member(conn, pid, country)
            linked += 1
    return linked


def scrape_fixtures(conn):
    """Live: read the WC 2026 schedule from FBref and write fixtures."""
    import soccerdata as sd
    fbref = sd.FBref(leagues="INT-World Cup", seasons="2026")
    sched = fbref.read_schedule().reset_index()
    sched.columns = [c if isinstance(c, str) else c[-1] for c in sched.columns]
    count = 0
    for r in sched.to_dict("records"):
        repo.upsert_fixture(conn, {
            "match_id": str(r.get("game_id") or r.get("game")),
            "date": str(r.get("date")),
            "home_country": r.get("home_team") or r.get("home"),
            "away_country": r.get("away_team") or r.get("away"),
            "stage": r.get("round") or r.get("stage"),
        })
        count += 1
    return count
```

SEED_BACKLOG = [
    "ML projections of player stats",
    "Bookmaker-odds ingestion and +EV value detection",
    "Real-time / in-play data",
]


def add_backlog_item(conn, title, notes="", now=None):
    cur = conn.execute(
        "INSERT INTO backlog (title, notes, status, created_at) "
        "VALUES (?, ?, 'open', ?)", (title, notes, now or ""))
    conn.commit()
    return cur.lastrowid


def list_backlog(conn):
    rows = conn.execute(
        "SELECT id, title, notes, status, created_at FROM backlog ORDER BY id")
    cols = ["id", "title", "notes", "status", "created_at"]
    return [dict(zip(cols, r)) for r in rows]


def complete_backlog_item(conn, item_id):
    conn.execute("UPDATE backlog SET status='done' WHERE id=?", (item_id,))
    conn.commit()


def seed_backlog(conn, now=None):
    existing = {i["title"] for i in list_backlog(conn)}
    for title in SEED_BACKLOG:
        if title not in existing:
            add_backlog_item(conn, title, now=now)

"""DB-backed helpers that feed the matchup engine. Read-only queries."""
from wcbets.analysis.normalise import per90_row
from wcbets.config import STAT_COLUMNS

_P90_KEYS = [c for c in STAT_COLUMNS if c != "minutes"]


def _row_to_player(row, cols):
    d = dict(zip(cols, row))
    return per90_row(d, _P90_KEYS)


def players_by_country(conn, country):
    cols = ["player_id", "name", "position"] + STAT_COLUMNS
    sql = f"""
        SELECT p.player_id, p.name, p.position, {', '.join('s.'+c for c in STAT_COLUMNS)}
        FROM players p
        JOIN national_squads ns ON ns.player_id = p.player_id
        JOIN player_stats s ON s.player_id = p.player_id
        WHERE ns.country = ?
    """
    return [_row_to_player(r, cols) for r in conn.execute(sql, (country,))]


def peer_distribution(conn, position, p90_key):
    base = p90_key[:-4] if p90_key.endswith("_p90") else p90_key
    cols = ["player_id", "name", "position"] + STAT_COLUMNS
    sql = f"""
        SELECT p.player_id, p.name, p.position, {', '.join('s.'+c for c in STAT_COLUMNS)}
        FROM players p JOIN player_stats s ON s.player_id = p.player_id
        WHERE p.position = ?
    """
    out = []
    for r in conn.execute(sql, (position,)):
        out.append(_row_to_player(r, cols)[p90_key])
    return out


def list_countries(conn):
    return [r[0] for r in conn.execute(
        "SELECT DISTINCT country FROM national_squads ORDER BY country")]

"""Streamlit dashboard: pick two countries, see matchup flags; manage backlog."""
import streamlit as st
from wcbets.config import DB_PATH, FLAG_PERCENTILE
from wcbets.db import repository as repo
from wcbets.analysis import matchup_query as mq
from wcbets.analysis.matchup import evaluate_matchup, RULES

st.set_page_config(page_title="WC Bets", layout="wide")
conn = repo.connect(DB_PATH)
repo.init_db(conn)
repo.seed_backlog(conn, now="seed")

st.title("WC Bets — matchup flags")
tab_matchups, tab_backlog = st.tabs(["Matchups", "Backlog"])

with tab_matchups:
    countries = mq.list_countries(conn)
    if not countries:
        st.info("No squads loaded yet. Run the scrapers or import CSVs first.")
    else:
        c1, c2 = st.columns(2)
        home = c1.selectbox("Defending side", countries, key="home")
        away = c2.selectbox("Attacking side", countries,
                            index=min(1, len(countries) - 1), key="away")
        defenders = [p for p in mq.players_by_country(conn, home)
                     if (p["position"] or "").startswith("D")]
        attackers = [p for p in mq.players_by_country(conn, away)
                     if (p["position"] or "").startswith(("F", "M"))]
        any_flag = False
        for d in defenders:
            for a in attackers:
                peers = {}
                for _, def_stat, atk_stat in RULES:
                    peers[f"defender_{def_stat}"] = mq.peer_distribution(
                        conn, d["position"], def_stat)
                    peers[f"attacker_{atk_stat}"] = mq.peer_distribution(
                        conn, a["position"], atk_stat)
                flags = evaluate_matchup(d, a, peers, threshold=FLAG_PERCENTILE)
                for f in flags:
                    any_flag = True
                    st.warning(
                        f"**{f['label']}** — {f['defender']} "
                        f"({f['defender_stat']}={f['defender_value']}, "
                        f"{int(f['defender_pct']*100)}th pct) vs {f['attacker']} "
                        f"({f['attacker_stat']}={f['attacker_value']}, "
                        f"{int(f['attacker_pct']*100)}th pct)")
        if not any_flag:
            st.success("No flagged matchups at the current threshold.")

with tab_backlog:
    st.subheader("Backlog / future work")
    new = st.text_input("Add an item")
    if st.button("Add") and new.strip():
        repo.add_backlog_item(conn, new.strip(), now="ui")
        st.rerun()
    for item in repo.list_backlog(conn):
        done = item["status"] == "done"
        label = f"~~{item['title']}~~" if done else item["title"]
        cols = st.columns([6, 1])
        cols[0].markdown(f"{label}  \n<small>{item['notes']}</small>",
                         unsafe_allow_html=True)
        if not done and cols[1].button("Done", key=f"done{item['id']}"):
            repo.complete_backlog_item(conn, item["id"])
            st.rerun()
